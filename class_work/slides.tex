% 3. Geometric mismatch
% =========================================================================
\begin{frame}{Trees grow exponentially. Euclidean space does not.}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.5\textwidth}
      \vspace{0.4em}
      \textbf{Euclidean ball, radius $r$:}\\
      volume grows as $r^d$ \emph{(polynomial)}.
      \vspace{0.8em}

      \textbf{Tree, depth $r$:}\\
      $\#$ nodes grows as $b^r$ \emph{(exponential)}.
      \vspace{0.8em}

      $\Rightarrow$ Embedding a deep tree into $\mathbb{R}^d$ \alert{must
      distort distances}.
      \vspace{0.8em}

      \textbf{Poincar\'e ball:} hyperbolic, exponential volume growth ---
      a natural home for trees \\\scriptsize{[Nickel \& Kiela 2017;
      Sala et al.\ 2018]}.
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.95]
        % Euclidean disk
        \draw[thick, eu] (0,0) circle (1.5);
        \node[eu] at (0,1.85) {\small Euclidean};
        \foreach \r in {0.5, 1.0, 1.5} {
          \draw[eu, very thin, dashed] (0,0) circle (\r);
        }
        \node[eu, font=\scriptsize] at (0,-1.9) {volume $\sim r^d$};

        % Hyperbolic disk
        \begin{scope}[xshift=4.2cm]
          \draw[thick, hy] (0,0) circle (1.5);
          \node[hy] at (0,1.85) {\small Hyperbolic};
          % concentric circles of equal hyperbolic distance bunch toward boundary
          \foreach \r in {0.7, 1.05, 1.27, 1.4, 1.46} {
            \draw[hy, very thin, dashed] (0,0) circle (\r);
          }
          \node[hy, font=\scriptsize] at (0,-1.9) {volume $\sim e^r$};
        \end{scope}
      \end{tikzpicture}
      \vspace{0.4em}

      {\scriptsize Equal-distance shells in each geometry. Hyperbolic
      shells crowd toward the boundary --- room for a tree's leaves.}
    \end{column}
  \end{columns}
\end{frame}

% 4. Research question
% =========================================================================
\begin{frame}{Research question}
  \vspace{0.3em}
  \begin{block}{}
    \large
    Do \Hy\ embeddings recover the WikiArt style hierarchy more
    faithfully than \Eu\ embeddings of matched dimension and training
    budget?
  \end{block}

  \vspace{0.7em}
  We answer along three axes:
  \begin{enumerate}
    \item \textbf{Scaling.} Sweep dimension $d \in \{2,4,8,16,32,64\}$
          and curvature $c \in \{0.1, 0.3, 1, 3\}$.
    \item \textbf{Training-time hierarchy.} Add a regularizer with
          weight $\lambda$ that pulls prototypes toward the tree.
    \item \textbf{Tree sensitivity.} Re-train against alternative
          reference trees, including a data-driven empirical one.
  \end{enumerate}
\end{frame}

% 6. Method: architecture
% =========================================================================
\begin{frame}{Method: prototype softmax in two geometries}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
      \textbf{Pipeline}
      \begin{itemize}\setlength\itemsep{0.15em}
        \item Frozen CLIP ViT-B/16 features ($\mathbb{R}^{512}$).
        \item 2-layer MLP $\to$ embedding $z \in \mathbb{R}^d$.
        \item Geometry head: identity \emph{or} $\exp_0^c$ onto $\mathbb{B}^d_c$.
        \item 27 learnable prototypes; logits $= -d(z, p_k)$.
      \end{itemize}
      \vspace{0.4em}
      \alert{Switching geometry changes only two things:} the head's
      final transform and the distance metric.
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}[
          every node/.style={font=\scriptsize, align=center},

% 7. Hierarchy-aware loss + training details
% =========================================================================
\begin{frame}{Hierarchy-aware loss \& training}
  \textbf{Loss.} Cross-entropy plus a regularizer pulling the prototype
  distance matrix toward the tree distance matrix:
  \[
    \mathcal{L} = \mathcal{L}_{CE} + \lambda \cdot
      \frac{1}{|\mathcal{P}|}\!\!\sum_{(i,j)\in\mathcal{P}}\!\!\!
      \left(\frac{d(p_i, p_j)}{\bar d} - \frac{T_{ij}}{\bar T}\right)^{\!2}.
  \]
  Mean-normalised --- constrains the \emph{shape} of $d(p_i, p_j)$, not its scale.

  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.55\textwidth}
      \textbf{Training} \scriptsize
      \begin{itemize}\setlength\itemsep{0pt}
        \item 150 runs $\times$ 3 seeds; 30 epochs; batch 4096.
        \item Adam $10^{-3}$ (head), Riemannian Adam $10^{-2}$ (\Hy\ prototypes).
        \item $<\!10$ s / run; full sweep $<\!30$ min on one M-series GPU.
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \textbf{Evaluation} \scriptsize
      \begin{itemize}\setlength\itemsep{0pt}
        \item \emph{Classification:} top-1, top-5, balanced acc.
        \item \emph{Global tree:} prototype/tree Spearman, distortion.
        \item \emph{Local hierarchy:} sibling \& cousin recall@$k$.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% 12. Conclusions

\begin{frame}{Takeaways \& future work}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.55\textwidth}
      \textbf{What we found}
      \begin{itemize}\setlength\itemsep{0.15em}
        \item \Hy\ wins \emph{local} hierarchy: sibling recall +4--5 pp.
              Robust across $\lambda$, $d$, reference tree, class weighting.
        \item \Eu\ wins \emph{classification}: top-1 +4--6 pp, equally robust.
        \item The \emph{global} tree gap depends on the reference ---
              flips in \Hy's favour against a data-driven tree.
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \textbf{Future work}
      \begin{itemize}\setlength\itemsep{0.15em}
        \item Fully hyperbolic head (M\"obius layers).
        \item Alternative regularizers: Spearman-rank, Sammon-style.
        \item Multiple empirical trees to test Phase~4's robustness.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
